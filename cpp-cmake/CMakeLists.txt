cmake_minimum_required(VERSION 3.20)
project(@PROJECT_NAME@ VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ENABLE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(ENABLE_PROFILING "Build with profiling-friendly options" OFF)
option(BUILD_TESTS "Build unit tests" @BUILD_TESTS_DEFAULT@)
add_library(${PROJECT_NAME}_lib src/app.cpp)
target_include_directories(${PROJECT_NAME}_lib PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_options(${PROJECT_NAME}_lib PRIVATE -Wall -Wextra -Wpedantic -Wshadow)
if(ENABLE_PROFILING)
  target_compile_options(${PROJECT_NAME}_lib PRIVATE -fno-omit-frame-pointer)
endif()
if(ENABLE_SANITIZERS)
  target_compile_options(${PROJECT_NAME}_lib PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(${PROJECT_NAME}_lib PRIVATE -fsanitize=address,undefined)
endif()
add_executable(${PROJECT_NAME} src/main.cpp)
target_link_libraries(${PROJECT_NAME} PRIVATE ${PROJECT_NAME}_lib)
if(ENABLE_PROFILING)
  target_compile_options(${PROJECT_NAME} PRIVATE -fno-omit-frame-pointer)
endif()
if(ENABLE_SANITIZERS)
  target_compile_options(${PROJECT_NAME} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(${PROJECT_NAME} PRIVATE -fsanitize=address,undefined)
endif()
if(BUILD_TESTS)
  enable_testing()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(CRITERION REQUIRED IMPORTED_TARGET criterion)
  add_executable(${PROJECT_NAME}_tests tests/test_app.cpp)
  target_link_libraries(${PROJECT_NAME}_tests PRIVATE ${PROJECT_NAME}_lib PkgConfig::CRITERION)
  add_test(NAME ${PROJECT_NAME}_tests COMMAND ${PROJECT_NAME}_tests)
endif()
